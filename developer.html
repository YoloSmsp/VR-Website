Quickstart Installation

Install the npm package or import via WebXR CDN

npm yarn
# Install Forma Horizon Spatial Engine
npm install @infinitev/spatial-sdk @infinitev/webxr-bridge

Initialize Spatial Tracking Engine

import { SpatialEngine, HapticChannel } from '@infinitev/spatial-sdk';

// 1. Initialize engine with 120Hz tracking loop
const engine = await SpatialEngine.init({
  fov: 'bionic-140',
  eyeTracking: true,
  motionToPhoton: 0.8,
});

// 2. Subscribe to hand skeletal tracking
engine.onHandPose((hand) => {
  const indexTip = hand.joints.get('INDEX_FINGER_TIP');
  if (hand.pinching) {
    engine.haptics.trigger(HapticChannel.INDEX, { intensity: 0.85, durationMs: 12 });
  }
});

// 3. Render frame to Micro-OLED dual canvas
engine.startRenderLoop((xrFrame) => {
  engine.renderSpatialScene(xrFrame);
});
